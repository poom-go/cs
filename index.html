// ============================================================ // 반품 양품화 관리 - Apps Script v2 // ============================================================ function doGet(e) { const action = (e.parameter && e.parameter.action) || 'cancelList'; let result; if (action === 'cancelList') result = getCancelList(); else if (action === 'loadSession') result = loadSession(e.parameter.key); else result = {ok: false, error: 'Unknown action: ' + action}; return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { let result; try { const data = JSON.parse(e.postData.contents); if (data.action === 'saveSession') result = saveSession(data); else result = saveReport(data); } catch(err) { result = {ok: false, error: err.toString()}; } return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // ── 마감후취소 B열 ───────────────────────────────────────── function getCancelList() { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('마감후취소'); if (!sheet) return {ok: false, error: '마감후취소 탭 없음'}; const lastRow = sheet.getLastRow(); if (lastRow < 2) return {ok: true, invoices: []}; const values = sheet.getRange(2, 2, lastRow - 1, 1).getValues(); const invoices = values .map(row => String(row[0]).trim()) .filter(v => v && v !== '' && v !== 'undefined' && v !== 'null'); return {ok: true, invoices: invoices}; } catch(err) { return {ok: false, error: err.toString()}; } } // ── 세션 저장 ───────────────────────────────────────────── function saveSession(data) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName('작업세션'); if (!sheet) { sheet = ss.insertSheet('작업세션'); sheet.appendRow(['세션키', '저장일시', 'RAW데이터', '작업상태']); sheet.getRange(1,1,1,4).setFontWeight('bold').setBackground('#f0f0f0'); } const key = data.key; if (!key) return {ok: false, error: 'key 없음'}; const rawJson = '[]'; const stateJson = JSON.stringify({ confirmations: data.confirmations || {}, edits: data.edits || {}, inboundNos: data.inboundNos || {}, noSendSet: data.noSendSet || [], cancelSet: data.cancelSet || [] }); const now = new Date().toLocaleString('ko-KR'); const allVals = sheet.getDataRange().getValues(); let rowIdx = -1; for (let i = 1; i < allVals.length; i++) { if (String(allVals[i][0]).trim() === String(key).trim()) { rowIdx = i + 1; break; } } if (rowIdx > 0) { sheet.getRange(rowIdx, 1, 1, 4).setValues([[key, now, rawJson, stateJson]]); } else { sheet.appendRow([key, now, rawJson, stateJson]); } return {ok: true}; } catch(err) { return {ok: false, error: err.toString()}; } } // ── 세션 불러오기 ───────────────────────────────────────── function loadSession(key) { try { if (!key) return {ok: false, error: 'key 없음'}; const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('작업세션'); if (!sheet) return {ok: true, data: null}; const allVals = sheet.getDataRange().getValues(); for (let i = 1; i < allVals.length; i++) { if (String(allVals[i][0]).trim() === String(key).trim()) { const state = JSON.parse(allVals[i][3] || '{}'); return {ok: true, data: {...state, savedAt: allVals[i][1]}}; } } return {ok: true, data: null}; } catch(err) { return {ok: false, error: err.toString()}; } } // ── 반품리포트 누적 ─────────────────────────────────────── function saveReport(data) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName('반품리포트'); if (!sheet) { sheet = ss.insertSheet('반품리포트'); sheet.appendRow(['발송일시','고객사명','CS번호','주문번호','원송장번호', '회수송장번호','바코드','SKU명','정상수량','소비기한', '로트','반송사유','입고번호','검품일']); sheet.getRange(1,1,1,14).setFontWeight('bold').setBackground('#f0f0f0'); } const rows = data.rows || []; rows.forEach(r => { sheet.appendRow([ new Date().toLocaleString('ko-KR'), r.company||'', r.csNo||'', r.orderNo||'', r.origInvoice||'', r.retInvoice||'', r.barcode||'', r.skuName||'', r.normalQty||0, r.expDate||'', r.lot||'', r.returnReason||'', r.inboundNo||'', r.inspDate||'' ]); }); return {ok: true, added: rows.length}; } catch(err) { return {ok: false, error: err.toString()}; } }
